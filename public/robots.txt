# Search engines may index the home page only. The game pages (?game=...)
# and the search and letter filters (?q=..., ?letter=...) are all query
# strings on that same address, so the "/$" rule below admits the bare home
# page and nothing else. The catalog editor (/studio/) is kept out too; its
# own page also says noindex.
#
# The site is drawn in the browser, so crawlers need the app's code and the
# catalog to render the home page: those stay allowed.

User-agent: *
Allow: /$
Allow: /assets/api/
Allow: /assets/app/
Allow: /assets/icons/
Allow: /manifest.webmanifest
Disallow: /
Disallow: /studio/
